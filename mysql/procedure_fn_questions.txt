CREATE DATABASE CollegeDB;

USE CollegeDB;

CREATE TABLE Student(
    sid INT PRIMARY KEY,
    sname VARCHAR(30),
    age INT,
    gender CHAR(1),
    city VARCHAR(20),
    marks INT,
    deptid INT
);

CREATE TABLE Department(
    deptid INT PRIMARY KEY,
    deptname VARCHAR(30),
    hod VARCHAR(30)
);

CREATE TABLE Employee(
    empid INT PRIMARY KEY,
    empname VARCHAR(30),
    salary DECIMAL(10,2),
    designation VARCHAR(30),
    deptid INT
);
